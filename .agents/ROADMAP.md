```javascript
/**
 * Aguarda que o último balão de mensagem enviado receba a confirmação do WhatsApp (1 risco = Enviado, 2 riscos = Entregue ou Lido)
 * @param {number} timeoutMs Tempo máximo de espera para confirmação (padrão 120000ms / 2 minutos)
 * @param {object} mochaContext Contexto do Mocha (this) do bloco it para acionar skip pontual se necessário
 */
aguardarConfirmacaoEntregaMensagem(timeoutMs = 120000, mochaContext = null) {
  const isMock = Cypress.env('USE_MOCKS') === true || Cypress.env('USE_MOCKS') === 'true';
  if (isMock) {
    cy.log('ℹ️ [Modo Mock]: Confirmação de entrega em tempo real ignorada em modo mock.');
    return;
  }

  cy.log(`⏳ [Aguardando Entrega/Envio]: Aguardando confirmação (✓ ou ✓✓) do WhatsApp por até ${timeoutMs / 1000}s...`);

  // 1. Garante que o indicador temporário de upload sumiu
  cy.contains('div, p, span', /Enviando mensagem/i, { timeout: 10000 }).should('not.exist');

  // 2. Sincronização resiliente no body aceitando 1 risco (Enviado), 2 riscos (Entregue) ou Lido
  cy.get('body', { timeout: timeoutMs }).should(($body) => {
    const $bubbles = $body.find('article[class*="_bubbleRowEnd_"], div[class*="_bubbleEmployee_"]');
    const $lastBubble = $bubbles.last();

    if ($lastBubble.length > 0 && $lastBubble[0].scrollIntoView) {
      $lastBubble[0].scrollIntoView({ behavior: 'smooth', block: 'end' });
    }

    // Detecção precoce de falha explícita no WhatsApp Gateway (evita queimar 120s inutilmente)
    const hasFailed = $lastBubble.find('span[aria-label*="Falha no envio"], span[class*="_statusIconError_"]').length > 0;
    if (hasFailed) {
      throw new Error('Falha no envio do anexo: O WhatsApp/Meta rejeitou o envio deste arquivo para o número do destinatário.');
    }

    const hasConfirmation = $lastBubble.find(
      'span[aria-label*="Enviad"], ' +
      'span[aria-label*="Entregu"], ' +
      'span[aria-label*="Lida"], ' +
      'span[aria-label*="Lido"], ' +
      'span[class*="_statusIcon_"]:has(svg path[d*="18 6"]), ' +
      'svg path[d*="18 6"]'
    ).length > 0;

    const isSending = $lastBubble.find('span[aria-label*="Aguardando"], span[aria-label*="Enviando"]').length > 0;

    expect(hasConfirmation && !isSending, 'Confirmação de envio/entrega (✓) do anexo visível no WhatsApp').to.be.true;
  });

  cy.log('✅ [Anexo Confirmado]: Confirmação de envio/entrega recebida do WhatsApp!');
}
```
